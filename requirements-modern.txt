# The portable stack: identical versions on Windows, macOS Apple Silicon and Linux.
#
# EXACT PINS ON PURPOSE. Unpinned, pip resolves differently on each machine and
# the two platforms drift apart — which defeats the point of having one app that
# behaves the same everywhere. These are the versions validated end-to-end on
# 2026-08-10: the released convlstm_ICA_12_train.h5 loads and scores real
# recordings, with no detection decision changing against the Python 3.6 build.
#
#   conda create -n seizmodern python=3.11
#   conda activate seizmodern
#   pip install -r requirements-modern.txt
#   pip install pyinstaller==6.22.0        # only needed to BUILD the app
#
# NOTE FOR ANYONE READING THIS WHO IS NOT BUILDING THE APP:
# none of this is needed to USE the application. The frozen build bundles its
# own Python and every library below. A clinician copies a folder and runs it.

# --- numeric core -------------------------------------------------------
# numpy MUST stay below 2: the `stft` package calls np.lib.pad, removed in 2.0.
numpy==1.26.4
scipy==1.17.1

# scikit-learn's version is no longer load-bearing: utils/fastica_pinned.py
# vendors the FastICA that MNE would otherwise call, because that dependency
# was what tied the pipeline to one machine. Pinned anyway so two platforms
# resolve to the same thing.
scikit-learn==1.9.0

# --- EEG / signal -------------------------------------------------------
mne==1.12.1
pyEDFlib==0.1.42
stft==0.5.2

# --- model --------------------------------------------------------------
# TF 2.x rather than the paper's 1.15, which has no Apple Silicon wheel and
# never will. Verified to agree with TF 1.15 to 6.8e-09 on real windows.
tensorflow==2.21.0
# Keras 2, which TF 2.16+ no longer bundles. Not optional: with Keras 3 the same
# weights and the same input give a different answer - measured 0.9779213071 vs
# 0.9663650393 on one window, and up to 0.44 across the corpus. The paper states
# Keras 2.0, so this is what reproduces it. models/convlstm_tf2.py sets
# TF_USE_LEGACY_KERAS=1 to select it.
tf-keras==2.21.0

# --- interface ----------------------------------------------------------
PyQt5==5.15.11
pyqtgraph==0.14.0

# --- required by MNE 1.x, not by the GUI --------------------------------
# MNE 0.19 did not need this. MNE 1.x imports mne.viz.topomap from
# mne.preprocessing.ica, so matplotlib became a hard runtime dependency of the
# ICA stage itself. A build that excludes it freezes cleanly and then dies on
# the first window scored.
matplotlib==3.11.1
